    function moveFocus(direction){
      const items=$$("[data-focusable]").filter(el=>!el.closest(".view:not(.active)")&&!el.closest(".side-rail:not(.open)")&&!el.disabled);
      const current=document.activeElement;if(!items.includes(current)){items[0]?.focus();return}
